async function payFlutterwave(){ const email = document .getElementById("deliveryEmail") .value .trim(); const msg = document .getElementById("payMsg"); if(!cart.length){ msg.textContent = "Your cart is empty."; return; } if(!email){ msg.textContent = "Enter your delivery email."; return; } if(!currentSession?.user){ msg.textContent = "Please log in to use your wallet."; openAuth("login"); return; } const amount = cart.reduce( (total, product) => total + Number(product[2]), 0 ); msg.textContent = "Checking wallet balance..."; try{ /* Get current wallet balance */ const wallet = await walletRequest({ action:"balance" }); const balance = Number(wallet.balance) || 0; if(balance < amount){ msg.textContent = "Insufficient wallet balance. Your balance is " + money(balance) + "."; return; } /* Buy each product using wallet */ const orders = []; for(const product of cart){ const result = await fetch( "/api/order", { method:"POST", headers:{ "Content-Type": "application/json" }, body: JSON.stringify({ action:"buy", user_id: currentSession.user.id, customer_email: email, product_name: product[1], category: product[0], amount: Number(product[2]) }) } ); const data = await result.json(); if(!result.ok || !data.success){ throw new Error( data.error || "Unable to complete purchase." ); } orders.push(data); } /* Purchase successful */ const newBalance = orders.length ? Number( orders[orders.length - 1].balance ) : balance; msg.textContent = "Purchase successful! Your wallet has been charged."; /* Clear cart */ cart = []; renderCart(); document .getElementById("drawer") .classList.remove("show"); /* Refresh wallet if account is open */ if( document .getElementById("accountModal") .classList .contains("show") ){ await loadWallet(); } /* Show order information */ const orderNumbers = orders .map( order => order.order_number ) .join(", "); setTimeout(() => { alert( "Purchase successful!\n\n" + "Order: " + orderNumbers + "\n" + "Email: " + email + "\n" + "Amount: " + money(amount) + "\n" + "Remaining balance: " + money(newBalance) ); },300); }catch(error){ console.error( "Wallet purchase error:", error ); msg.textContent = error.message || "Wallet purchase failed."; } }
